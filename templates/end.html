---------------- HELPER FUNCTIONS ---------------- def final_battle(with_eleven): """Apply final battle logic (like CLI scene5)""" health = session["health"] allies = session["allies"] if with_eleven and allies.get("Eleven", False) and health > 50: # Best ending session["ending"] = "best" elif with_eleven and health > 20: # Neutral ending session["health"] -= 30 session["ending"] = "neutral" else: # Bad ending session["health"] -= 50 session["ending"] = "bad" #
